	on_weekly_EQS = {
		effect = {
			log = "[GetDateText]: ROOT: [Root.GetName] on_weekly_EQS"
			if = {
				limit = {
					has_country_flag = greatwar_darkwar_enabled
				}
				add_to_variable = { global.CHN_infiltration_level = global.CHN_infiltration_weekly_change }
				clamp_variable = {
					var = global.CHN_infiltration_level
					min = 0
					max = 100
				}
				add_to_variable = { global.EQS_infiltration_level = global.EQS_infiltration_weekly_change }
				clamp_variable = {
					var = global.EQS_infiltration_level
					min = 0
					max = 100
				}
				EQS = {
					set_temp_variable = { EQS_DW_research_surrender_temp = global.CHN_infiltration_level }
					multiply_temp_variable = { EQS_DW_research_surrender_temp = -0.005 }
					multiply_temp_variable = { EQS_DW_research_surrender_temp = 0.5 } #For some reason the math only works if I multiply twice
					set_variable = { EQS_DW_research_speed = EQS_DW_research_surrender_temp }
					set_variable = { EQS_DW_surrender_limit = EQS_DW_research_surrender_temp }
					
					set_temp_variable = { EQS_DW_stability_ws_temp = global.CHN_infiltration_level }
					multiply_temp_variable = { EQS_DW_stability_ws_temp = -0.005 }
					set_variable = { EQS_DW_stability_factor = EQS_DW_stability_ws_temp }
					set_variable = { EQS_DW_war_support_factor = EQS_DW_stability_ws_temp }
				}
				if = {
					limit = {
						check_variable = { global.CHN_infiltration_level > 50 }
					}
					EQS = {
						set_temp_variable = { EQS_DW_landfort_defence_temp = global.CHN_infiltration_level }
						subtract_from_temp_variable = { EQS_DW_landfort_defence_temp = 50 }
						multiply_temp_variable = { EQS_DW_landfort_defence_temp = -0.005 }
						set_variable = { EQS_DW_land_bunker_effectiveness_factor = EQS_DW_landfort_defence_temp }
						set_variable = { EQS_DW_army_defence_factor = EQS_DW_landfort_defence_temp }
					}
				}
				else = {
					EQS = {
						set_variable = { EQS_DW_land_bunker_effectiveness_factor = 0 }
						set_variable = { EQS_DW_army_defence_factor = 0 }
					}
				}
				CHN = {
					set_temp_variable = { CHN_freerepair_temp = global.EQS_infiltration_level }
					multiply_temp_variable = { CHN_freerepair_temp = -0.01 }
					set_variable = { CHN_DW_industry_free_repair_factor = CHN_freerepair_temp }
					
					set_temp_variable = { CHN_everything_else_temp = global.EQS_infiltration_level }
					multiply_temp_variable = { CHN_everything_else_temp = 0.005 }
					set_variable = { CHN_DW_required_garrison_factor = CHN_everything_else_temp }
					set_variable = { CHN_DW_resistance_activity = CHN_everything_else_temp }
					set_variable = { CHN_DW_resistance_target = CHN_everything_else_temp }
					multiply_temp_variable = { CHN_everything_else_temp = -1 }
					set_variable = { CHN_DW_compliance_growth = CHN_everything_else_temp }
				}
			}
			if = {
				limit = {
					has_country_flag = EQS_HIP_elements_sent
					has_war = yes
				}
				clr_country_flag = EQS_HIP_elements_sent
				country_event = hippogriff_abyss_resistance.19
			}
			if = {
				limit = {
					has_government = democratic
					has_war_with = NLR
					has_volunteers_amount_from = {
						tag = JUN
						count > 9
					}
					NOT = {
						has_country_flag = JUN_troops_arrived
					}
				}
				set_country_flag = JUN_troops_arrived
			}
			if = { # AF26
				limit = {
					has_country_flag = EQS_ED_harmony_magic_enabled
				}
				add_to_variable = { EQS_ED_friendship_power_current = EQS_ED_friendship_power_gain }
				clamp_variable = { var = EQS_ED_friendship_power_current min = 0 max = 100 }
			}
			if = { # AF26
				limit = {
					has_dynamic_modifier = {
						modifier = EQS_ED_centre_for_artefact_research
					}
				}
				ED_recalculate_centre_for_artefact_research_vars = yes
			}
			if = { # AF26
				limit = {
					has_dynamic_modifier = {
						modifier = ED_magic_balance_effect
					}
				}
				ED_decay_magical_balance_effect = yes 
			}
		}
	}

	on_peaceconference_ended = {
		effect = {
			# PUT EVERYTHING IN THE IF BELOW
			# THE bypass_on_peaceconference_ended FLAG **HAS** TO BE SET BEFORE CALLING THE white_peace COMMAND AND CLEARED AFTERWARDS
			if = {
				limit = {
					NOT = {
						has_country_flag = bypass_on_peaceconference_ended
						FROM = { has_country_flag = bypass_on_peaceconference_ended }
					}
				}
				log = "[GetDateText]: ROOT: [Root.GetName] FROM: [From.GetName] on_peaceconference_ended"
				if = {
					limit = {
						has_global_flag = equestrian_civil_war
						NOT = { has_global_flag = equestrian_civil_war_ended }
						FROM = {
							OR = {
								original_tag = EQS
								tag = NLR
							}
						}
					}
					set_global_flag = equestrian_civil_war_ended
				}
				if = {
					limit = {
						is_eqs_civil_war_tag = yes
						NOT = {
							original_tag = EQS
							original_tag = NLR
						}
						FROM = { is_eqs_civil_war_tag = yes }
					}
					set_country_flag = { flag = bypass_annex_compliance days = 1 value = 1 }
					annex_country = { target = FROM }
				}
			}
		}
	}
